pw_mining_director = {
	category = ruler
	swappable_data = {
		default = {
			condition_string = RULER_JOB_TRIGGER
			building_icon = pw_building_erebus_project
			# clothes_texture_index = 2
			icon = miner
		}
	}
	
	tags = { minerals }

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_ruler_job

	possible = {
		# ruler_job_check_trigger = yes
	}

	resources = {
		category = planet_miners
		produces = {
			minerals = @pw_job_director_resource_base
		}
		upkeep = {
			consumer_goods = @pw_job_director_resource_upkeep
		}
	}

	planet_modifier = {
		planet_jobs_minerals_produces_mult = @pw_job_director_resource_mult
	}

	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_add = @pw_job_director_resource_amenities
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}

	weight = {
		weight = @ruler_job_weight
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|planet_jobs_minerals_produces_mult|
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		modifier = {
			factor = 0.2
			has_citizenship_rights = no
			NOT = { has_trait = trait_mechanical }
		}
		modifier = {
			factor = 2
			planet = {
				OR = {
					has_designation = col_mining
					has_designation = col_habitat_mining
				}
			}
		}
		modifier = {
			factor = 0.01
			exists = planet
			exists = planet.controller
			planet.controller = {
				OR = {
					is_country_type = swarm
					is_country_type = ai_empire
				}
			}
		}
	}
}
